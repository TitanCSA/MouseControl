cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(MouseControl VERSION 1.0.1 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with Visual Studio 2022 C++ tools on Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /PDBALTPATH:%_PDB%)
option(MOUSECONTROL_ASAN "AddressSanitizer tests; not for distribution" OFF)
if(MOUSECONTROL_ASAN)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  add_compile_options(/fsanitize=address /Zi)
  add_link_options(/DEBUG /INCREMENTAL:NO)
endif()
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_library(InterceptorSDK STATIC Interceptor/Interceptor.cpp)
target_include_directories(InterceptorSDK PUBLIC Interceptor)
target_link_libraries(InterceptorSDK PUBLIC setupapi newdev cfgmgr32 hid advapi32 user32 comctl32 ole32)
add_library(MouseCore STATIC App/Core.cpp)
target_include_directories(MouseCore PUBLIC App)
add_library(MouseRuntime STATIC App/Engine.cpp)
target_link_libraries(MouseRuntime PUBLIC MouseCore InterceptorSDK)
add_library(MouseUI STATIC App/UI.cpp App/Storage.cpp)
target_link_libraries(MouseUI PUBLIC MouseRuntime d2d1 dwrite windowscodecs dwmapi comdlg32 shell32 bcrypt uuid)
add_executable(MouseControl WIN32 App/Main.cpp App/App.rc)
target_link_libraries(MouseControl PRIVATE MouseUI)
target_link_options(MouseControl PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_SOURCE_DIR}/App/App.manifest")
enable_testing()
add_executable(MouseCoreTests App/Tests.cpp)
target_link_libraries(MouseCoreTests PRIVATE MouseCore)
add_test(NAME SequenceAndTriggers COMMAND MouseCoreTests)
add_executable(MouseRuntimeTests App/RuntimeTests.cpp)
target_link_libraries(MouseRuntimeTests PRIVATE MouseRuntime)
add_test(NAME DriverBoundary COMMAND MouseRuntimeTests)
add_test(NAME Interface COMMAND MouseControl --self-test "${CMAKE_BINARY_DIR}/ui-test.json")
set_tests_properties(SequenceAndTriggers DriverBoundary Interface PROPERTIES TIMEOUT 20)
